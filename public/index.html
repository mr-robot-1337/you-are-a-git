    git branch <name>Create a new branch
    git checkout <name>Switch branch (or -b)
    git merge <branch>Merge branch into HEAD
    git reset --hard HEAD~1Move branch back 1 commit
    git revert HEADCreate inverse commit
    git logShow commit history
